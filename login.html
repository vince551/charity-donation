// js/users.js import { db } from './firebase.js'; import { collection, getDocs } from "https://www.gstatic.com/firebasejs/10.8.0/firebase-firestore.js"; export async function loadUsers() { const container = document.getElementById('usersList'); if (!container) return; container.innerHTML = 'Loading...'; try { const snapshot = await getDocs(collection(db, "users")); container.innerHTML = ''; if (snapshot.empty) { container.innerHTML = 'No registered users found.'; return; } snapshot.forEach(docSnap => { const data = docSnap.data(); const row = document.createElement('tr'); row.innerHTML = ` ${data.email || 'N/A'} ${data.role || 'user'} `; container.appendChild(row); }); } catch (err) { container.innerHTML = 'Error loading users.'; } }
