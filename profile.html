// js/notifications.js export function showNotification(message, type = 'info') { const toast = document.createElement('div'); toast.className = `toast toast-${type}`; toast.innerText = message; toast.style.cssText = ` position: fixed; bottom: 20px; right: 20px; padding: 12px 20px; background: #10b981; color: white; border-radius: 6px; box-shadow: 0 4px 6px rgba(0,0,0,0.1); z-index: 9999; `; document.body.appendChild(toast); setTimeout(() => toast.remove(), 3000); }
